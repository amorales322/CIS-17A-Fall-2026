Function Board* newBord(fstream Ref gmeFile, Constant bool frmFile)
    Call gmeFile.seekg(0, ios::beg)
    Declare Board* board = new Board
    Set board->whoTurn = plyrClr::WHITE
    Set board->check = PlyrClr::NONE
    Set board->chkmte = PlyrClr::NONE
    Set board->surndr = PlyrClr::NONE
    Set board->exitGme = PlyrClr::NONE
    Set board->stlemte = false
    Set board->deadPos = false

    If fromFile Then
        Call read(gmeFile, board->whoTurn)
        Call read(gmeFile, board->check)
        Call read(gmeFile, board->chkmte)
        Call read(gmeFile, board->surndr)
        Call read(gmeFile, board->stlemte)
        Call read(gmeFile, board->deadPos)
        Set board->board = new Piece **[board->VERTSZE]

        For size_t row = 0 To board->VERTSZE
            Set board->board[row] = new Piece *[board->HORZNSZ]
            For size_t col = 0 To board->HORZNSZ
                Set board->board[row][col] = newPce(gmeFile)
            End For
        End For

        Set board->wKngPos = newCord(gmeFile)
        Set board->bKngPos = newCord(gmeFile)
    Else
        Call alocPce(board)
        Set board->wKngPos = newCord(8, 5)
        Set board->bKngPos = newCord(1, 5)
    End If

    Return board
End Function

Module toFile(fstream Reg gmeFile, fstream Ref plyr1, fstream Ref plyr2, Constant Board* Constant board)
    Call toFile(plyr1, board->white)
    Call toFile(plyr2, board->black)

    Call gmeFile.seekp(0, ios::beg)
    Call write(gmeFile, board->whoTurn)
    Call write(gmeFile, board->check)
    Call write(gmeFile, board->chkmte)
    Call write(gmeFile, board->surndr)
    Call write(gmeFile, board->stlemte)
    Call write(gmeFile, board->deadPos)

    For size_t row = 0 To board->VERTSZE
        For size_t col = 0 To board->HORZNSZ
            Call toFile(gmeFile, board->board[row][col])
        End For
    End For

    Call toFile(gmeFile, board->wKngPos)
    Call toFile(gmeFile, board->bKngPos)
End Module

Module rstBrd(Board* Constant board)
    Call dealloc(board->wKngPos)
    Call dealloc(board->bKngPos)
    Call dealloc(board->board, board->VERTSZE, board->HORZNSZ)

    Call alocPce(board)
    Set board->wKngPos = newCord(8, 5)
    Set board->bKngPos = newCord(1, 5)

    Set board->whoTurn = PlyrClr::WHITE
    Set board->check = PlyrClr::NONE
    Set board->chkmte = PlyrClr::NONE
    Set board->surndr = PlyrClr::NONE
    Set board->exitGme = PlyrClr::NONE
    Set board->stlemte = false
    Set board->deadPos = false
End Module

Module alocPce(Board* Constant board)
    Set board->board = new Piece **[board->VERTSZE]

    For size_t row = 0 To board->VERTSZE
        Set board->board[row] = new Piece *[board->HORZNSZ]
        For size_t col = 0 To board->HORZNSZ
            Set board->board[row][col] = nullptr
        End For
    End For

    For size_t col = 1 To 9
        Set board->board[2][col] = newPce(PlyrClr::BLACK, PieceID::PAWN)
        Set board->board[7][col] = newPce(PlyrClr::WHITE, PieceID::PAWN)
    End For

    //  King Initialization
    Set board->board[1][5] = newPce(PlyrClr::BLACK, PieceID::KING)
    Set board->board[8][5] = newPce(PlyrClr::WHITE, PieceID::KING)

    //  Queen Initialization
    Set board->board[1][4] = newPce(PlyrClr::BLACK, PieceID::QUEEN)
    Set board->board[8][4] = newPce(PlyrClr::WHITE, PieceID::QUEEN)

    //  Rook Initialization
    Set board->board[1][1] = newPce(PlyrClr::BLACK, PieceID::ROOK)
    Set board->board[1][8] = newPce(PlyrClr::BLACK, PieceID::ROOK)
    Set board->board[8][1] = newPce(PlyrClr::WHITE, PieceID::ROOK)
    Set board->board[8][8] = newPce(PlyrClr::WHITE, PieceID::ROOK)

    //  Bishop Initialization
    Set board->board[1][3] = newPce(PlyrClr::BLACK, PieceID::BISHOP)
    Set board->board[1][6] = newPce(PlyrClr::BLACK, PieceID::BISHOP)
    Set board->board[8][3] = newPce(PlyrClr::WHITE, PieceID::BISHOP)
    Set board->board[8][6] = newPce(PlyrClr::WHITE, PieceID::BISHOP)


    //  Knight Initialization
    Set board->board[1][2] = newPce(PlyrClr::BLACK, PieceID::KNIGHT)
    Set board->board[1][7] = newPce(PlyrClr::BLACK, PieceID::KNIGHT)
    Set board->board[8][2] = newPce(PlyrClr::WHITE, PieceID::KNIGHT)
    Set board->board[8][7] = newPce(PlyrClr::WHITE, PieceID::KNIGHT)

    For size_t row = 0 To board->VERTSZE
        For size_t col = 0 To board->HORZNSZ
            If board->board[row][col] == nullptr Then
                Set board->board[row][col] = newPce()
            End If
        End For
    End For
End Module

Function bool deadPos(Constant Board* Constant board)
    For size_t row = 1 To board->VERTSZE - 1
        For size_t col = 1 To board->HORZNSZ - 1
            If (curPce->pieceID != PieceID::KING) AND (curPce->pieceID != PieceID::BISHOP) AND (curPce->pieceID != PieceID::KNIGHT) Then
                Return false
            End If
        End For
    End For

    Return true
End Function

Module drawBrd(Constant Board* Constant board)
    Declare Piece* curPce
    Declare bool lghtCel

    Display "\n\n", board->black->name, "\n\n"
    For size_t cellRow = 3 To 9
        For size_t col 0 = To board->HORZNSZ - 1
            Select col
                Case 1:
                    Call drawA(cellRow)
                Case 2:
                    Call drawB(cellRow)
                Case 3:
                    Call drawC(cellRow)
                Case 4:
                    Call drawD(cellRow)
                Case 5:
                    Call drawE(cellRow)
                Case 6:
                    Call drawF(cellRow)
                Case 7:
                    Call drawG(cellRow)
                Case 8:
                    Call drawH(cellRow)
                Default:
                    Display getBlnk(24, false)
            End Select
        End For
        Display '\n'
    End For

    For size_t row = 1 To board->VERTSZE - 1
        Display getBlnk(24, false)
        For size_t col = 1 To board->HORZNSZ - 1
            Display '+', string(23, '-')
        End For
        Display "+\n"
        For size_t cellRow = 0 To 11
            For size_t col = 0 to board->HORZNSZ - 1
                If col == 0 Then
                    Select row
                        Case 1:
                            Call draw1(cellRow)
                        Case 2:
                            Call draw2(cellRow)
                        Case 3:
                            Call draw3(cellRow)
                        Case 4:
                            Call draw4(cellRow)
                        Case 5:
                            Call draw5(cellRow)
                        Case 6:
                            Call draw6(cellRow)
                        Case 7:
                            Call draw7(cellRow)
                        Case 8:
                            Call draw8(cellRow)
                        Default:
                            Display getBlnk(24, false)
                    End Select
                Else
                    Set curPce = getPce(board, row, col)
                    Set lghtCel = (col + row - 1) MOD 2 != 0
                    If curPce->empty != true Then
                        Select curPce->pieceI
                            Case PieceID::KING
                                Call drwKing(lghtCel, cellRow, curPce->pceClr)
                            Case PieceID::QUEEN
                                Call drwQuen(lghtCel, cellRow, curPce->pceClr)
                            Case PieceID::ROOK
                                Call drwRook(lghtCel, cellRow, curPce->pceClr)
                            Case PieceID::BISHOP
                                Call drwBshp(lghtCel, cellRow, curPce->pceClr)
                            Case PieceID::KNIGHT
                                Call drwKnit(lghtCel, cellRow, curPce->pceClr)
                            Case PieceID::PAWN:
                                Call drwPawn(lghtCel, cellRow, curPce->pceClr)
                            Default:
                                Call drwBlRw(lghtCel)
                        End Select
                    Else
                        Call drwBlRw(lghtCel)
                    End If
                End If
            End For
            Display "|\n"
        End For
    End For
    
    Display getBlnk(24, false)
    For size_t col = 1 To board->HORZNSZ - 1
        Display '+', string(23, '-')
    End For
    Display "+\n\n", board->white->name, "\n\n"
End Module

Module dealloc(Board* board)
    Call dealloc(board->board, board->VERTSZE, board->HORZNSZ)
    
    Call dealloc(board->wKngPos)
    Set board->wKngPos = nullptr
    
    Call dealloc(board->bKngPos)
    Set board->bKngPos = nullptr
    
    Call dealloc(board->white)
    Set board->white = nullptr
    
    Call dealloc(board->black)
    Set board->black = nullptr
    
    delete board
    Set board = nullptr
End Module

Module (dealloc Piece*** pceArry, Constant size_t vertSze, Constant size_t horzSze)
    If pceArry != nullptr Then
        For size_t row = 0 To vertSze
            For size_t col To horzSze
                Call dealloc(pceArry[row][col])
            End For
            delete []pceArry[row]
            Set pceArry[row] = nullptr
        End For
    End If
End Module

Function Coord* getInpt()
    Declare string input
    Input input

    If input == "-1" Then
        Return newCord(-1, -1)
    End If
    If input == "-2" Then
        Return newCord(-2, -2)
    End If
    If (input.length() = 2) AND (toupper(input[0]) <= 'H') AND (toupper(input[0]) >= 'A') AND (input[1] <= '8') AND (input[1] >= '1') Then
        Return fmChsNt(input)
    End If

    Display "Invalid board coordinate. Enter new board coordinate: "
    Return getInpt()
End Function

Function bool getOpt()
    Declare char opt
    Input opt
    Set opt = static_cast<char>(toupper(opt))

    If opt == 'Y' OR opt == 'N' Then
        Return opt == 'Y'
    End If

    Display "Invalid option, enter new option: "
    Return getOpt()
End Function

Module mvePce(Constant Board* Constant board, Constant Coord* Constant a, Constant Coord* Constant b)
    Declare Piece* pceA = getPce(board, a)
    Declare Piece* pceB = getPce(board, b)
    Declare Piece* tmp = pceA

    If pceB->empty != true Then
        If pceA->pceClr == PlyrClr::WHITE Then
            Display board->white->name, " has captured ", board->black->name, "'s ", getName(board, b), "!\n"
        Else
            Display board->black->name, " has captured ", board->white->name, "'s ", getName(board, b), "!\n"
        End If

        Call dealloc(board->board[b->row][b->col])
        Set board->board[b->row][b->col] = newPce()
        Set pceB = getPce(board, b)
    End If

    Set board->board[a->row][a->col] = pceB
    Set board->board[b->row][b->col] = tmp
End Module

Function Piece* getPce(Constant Board* Constant board, Constant Coord* Constant coord)
    Return getPce(board, coord->row, coord->col)
End Function

Function Piece* getPce(Constant Board* Constant board, Constant size_t row, Constant size_t col)
    Return board->board->[row][col]
End Function

Function Constant char* getName(Constant Board* Constant board, Constant Coord* Constant coord, Constant bool captlze)
    Return getName(board, coord->row, coord->col, captlze)
End Function

Function Constant char* getName(Constant Board* Constant board, Constant size_t row, Constant size_t col, Constant bool captlze)
    Declare Static Constant char* PCENME_U[7] = {"N/A", "King", "Queen", "Rook", "Bishop", "Knight", "Pawn"}
    Declare Static Constant char* PCENME_L[7] = {"N/A", "king", "queen", "rook", "bishop", "knight", "pawn"}

    If captlze Then
        Return PCENME_U[static_cast<size_t>(getPce(board, row, col)->pieceID)]
    End If

    Return PCENME_L[static_cast<size_t>(getPce(board, row, col)->pieceID)]
End Function

Function bool isOOB(Constant Coord* Constant coord)
    Return isOOB(coord->row, coord->col)
End Function

Function bool isOOB(const size_t row, const size_t col)
    Return row < 1 OR row > 8 OR col < 1 OR col > 8
End Function
